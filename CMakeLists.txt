cmake_minimum_required(VERSION 3.24)
project(Aurex VERSION 0.4.2 LANGUAGES C CXX)
set(AUREX_VERSION_SUFFIX "" CACHE STRING "Optional preview suffix, for example -beta.1")
if(NOT AUREX_VERSION_SUFFIX STREQUAL "" AND NOT AUREX_VERSION_SUFFIX MATCHES "^-(alpha|beta|rc)(\\.(0|[1-9][0-9]*))?$")
  message(FATAL_ERROR "Use a preview suffix such as -beta.1, or leave AUREX_VERSION_SUFFIX empty for a stable build")
endif()
set(AUREX_VERSION "${PROJECT_VERSION}${AUREX_VERSION_SUFFIX}")
file(WRITE "${PROJECT_BINARY_DIR}/release-version.txt" "${AUREX_VERSION}\n")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>")
endif()
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
option(HIVE_BUILD_DESKTOP "Build the native desktop application" ON)
option(BUILD_TESTING "Build local test suites (not distributed)" OFF)
option(HIVE_RELEASE_LTO "Optimize release builds across translation units" ON)
if(HIVE_RELEASE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(LANGUAGES C CXX)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
include(CTest)

find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
find_package(Threads REQUIRED)
if(HIVE_BUILD_DESKTOP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Concurrent Network WebSockets)
  if(WIN32)
    get_target_property(qt_linkage Qt6::Core TYPE)
    if(NOT qt_linkage STREQUAL "STATIC_LIBRARY")
      message(FATAL_ERROR "Windows builds require static Qt. Run build.cmd to build the dependencies.")
    endif()
  endif()
  if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
  endif()
endif()
include(cmake/Lua.cmake)

add_subdirectory(src/core/runtime)
add_subdirectory(src/coordination)
add_subdirectory(src/game)
add_subdirectory(src/platform)
add_subdirectory(src/game/actions)
add_subdirectory(src/core/scripting)
add_subdirectory(src/core/session)
add_subdirectory(src/core/module_host)
add_subdirectory(src/modules)
if(HIVE_BUILD_DESKTOP)
  add_subdirectory(src/reporting)
  add_subdirectory(src/app)
endif()
include(cmake/CheckTargets.cmake)

if(BUILD_TESTING AND EXISTS "${PROJECT_SOURCE_DIR}/tests/CMakeLists.txt")
  add_subdirectory(tests)
endif()
