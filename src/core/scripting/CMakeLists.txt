set(HIVE_MOVEMENT_CONTENT "#pragma once\n#include <map>\n#include <string>\nnamespace hive::movement {\ninline const std::map<std::string,std::string> bundled_movement = {\n")
foreach(recipe IN LISTS HIVE_MOVEMENT_FILES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${recipe}")
  get_filename_component(recipe_name "${recipe}" NAME_WE)
  file(READ "${recipe}" recipe_source)
  string(APPEND HIVE_MOVEMENT_CONTENT "{\"${recipe_name}\",R\"movement(${recipe_source})movement\"},\n")
endforeach()
string(APPEND HIVE_MOVEMENT_CONTENT "};\n}\n")
file(WRITE "${HIVE_MOVEMENT_HEADER}.in" "${HIVE_MOVEMENT_CONTENT}")
